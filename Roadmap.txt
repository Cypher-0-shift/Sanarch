Full roadmap + file structure

Phase 0 — Services to set up (do this today, in order)
1. Railway — railway.app — create account, install CLI (npm i -g @railway/cli), this hosts your FastAPI backend + PostgreSQL + Redis + Celery workers all in one project.
2. Anthropic API — console.anthropic.com — create account, generate API key, this is your OCR + NER engine.
3. Backblaze B2 — backblaze.com/b2 — create account, create a private bucket called sanarch-docs, generate application key with read/write access. Copy: Key ID, Application Key, bucket name, endpoint URL.
4. Firebase — console.firebase.google.com — create project sanarch, enable Authentication → Phone, download google-services.json for Android and GoogleService-Info.plist for iOS. Also enable Firebase App Distribution for sending test APKs.
5. ClamAV — runs as a Docker container inside your Railway project, no external account needed.

# services/extraction.py

import json
import httpx
from surya.ocr import run_ocr
from surya.model.detection.model import load_model as load_det_model
from surya.model.detection.processor import load_processor as load_det_processor
from surya.model.recognition.model import load_model as load_rec_model
from surya.model.recognition.processor import load_processor as load_rec_processor
import spacy
from app.config import settings

# Load models once at startup — not per request
det_model, det_processor = load_det_model(), load_det_processor()
rec_model, rec_processor = load_rec_model(), load_rec_processor()
nlp_med7 = spacy.load("en_core_med7_lg")  # after: pip install https://huggingface.co/kormilitzin/en_core_med7_lg/...

GROQ_URL = "https://api.groq.com/openai/v1/chat/completions"

STRUCTURE_PROMPT = """You are a medical document parser.
Given this raw extracted text and medical entities, return ONLY valid JSON. No explanation.

Schema:
{
  "document_type": "lab_report|prescription|hospital_summary|discharge|xray|other",
  "document_date": "YYYY-MM-DD or null",
  "hospital_name": "string or null",
  "doctor_name": "string or null",
  "patient_name": "string or null",
  "diagnosis": ["list"],
  "medications": [{"name": "", "dose": "", "frequency": "", "duration": ""}],
  "lab_values": [{"test_name": "", "value": "", "unit": "", "flag": "normal|high|low|null"}],
  "follow_up_date": "YYYY-MM-DD or null",
  "summary": "2 sentence plain English summary"
}"""


def ocr_with_surya(image_bytes: bytes) -> tuple[str, float]:
    """Returns (extracted_text, confidence_score)."""
    from PIL import Image
    import io
    image = Image.open(io.BytesIO(image_bytes))
    predictions = run_ocr(
        [image], [["en"]],
        det_model, det_processor,
        rec_model, rec_processor
    )
    lines = []
    confidences = []
    for page in predictions:
        for line in page.text_lines:
            lines.append(line.text)
            confidences.append(line.confidence)
    text = "\n".join(lines)
    avg_confidence = sum(confidences) / len(confidences) if confidences else 0
    return text, avg_confidence


def extract_entities_med7(text: str) -> dict:
    """Run Med7 NER on extracted text."""
    doc = nlp_med7(text)
    entities = {}
    for ent in doc.ents:
        label = ent.label_.lower()
        if label not in entities:
            entities[label] = []
        entities[label].append(ent.text)
    return entities


async def structure_with_groq(text: str, entities: dict) -> dict:
    """Use Groq free tier (Llama 3.1 70B) to structure into JSON."""
    user_content = f"""Raw OCR text:
{text[:3000]}

Med7 extracted entities:
{json.dumps(entities, indent=2)}

Return the JSON schema now."""

    async with httpx.AsyncClient() as client:
        response = await client.post(
            GROQ_URL,
            headers={
                "Authorization": f"Bearer {settings.groq_api_key}",
                "Content-Type": "application/json",
            },
            json={
                "model": "llama-3.1-70b-versatile",
                "messages": [
                    {"role": "system", "content": STRUCTURE_PROMPT},
                    {"role": "user", "content": user_content},
                ],
                "temperature": 0.1,
                "max_tokens": 1024,
            },
            timeout=30,
        )
        raw = response.json()["choices"][0]["message"]["content"].strip()
        if raw.startswith("```"):
            raw = raw.split("```")[1]
            if raw.startswith("json"):
                raw = raw[4:]
        return json.loads(raw)


async def extract_from_image(image_bytes: bytes, mime_type: str) -> dict:
    """Full pipeline: Surya OCR → Med7 NER → Groq structuring."""

    # Step 1: OCR
    if mime_type == "application/pdf":
        # Convert PDF pages to images first
        image_bytes = pdf_first_page_to_image(image_bytes)

    text, confidence = ocr_with_surya(image_bytes)

    # Fallback to Azure DI if Surya confidence is low
    if confidence < 0.70:
        text = await ocr_with_azure_fallback(image_bytes)

    # Step 2: NER
    entities = extract_entities_med7(text)

    # Step 3: Structure into JSON via Groq
    try:
        structured = await structure_with_groq(text, entities)
    except (json.JSONDecodeError, KeyError):
        # If Groq output is bad JSON, retry once with a stricter prompt
        structured = await structure_with_groq(
            text, entities  # you can add "IMPORTANT: return ONLY the JSON object" prefix
        )

    return structured


def pdf_first_page_to_image(pdf_bytes: bytes) -> bytes:
    """Convert first page of PDF to PNG bytes for OCR."""
    import fitz  # PyMuPDF — pip install pymupdf
    doc = fitz.open(stream=pdf_bytes, filetype="pdf")
    page = doc.load_page(0)
    pix = page.get_pixmap(dpi=200)
    return pix.tobytes("png")


async def ocr_with_azure_fallback(image_bytes: bytes) -> str:
    """Azure Document Intelligence fallback for low-confidence scans."""
    from azure.ai.formrecognizer import DocumentAnalysisClient
    from azure.core.credentials import AzureKeyCredential
    import io
    client = DocumentAnalysisClient(
        endpoint=settings.azure_di_endpoint,
        credential=AzureKeyCredential(settings.azure_di_key),
    )
    poller = client.begin_analyze_document(
        "prebuilt-read", document=io.BytesIO(image_bytes)
    )
    result = poller.result()
    return "\n".join([line.content for page in result.pages for line in page.lines])

Step 6 — Virus scan service
services/virus_scan.py:

import socket
from app.config import settings

def scan_bytes(file_bytes: bytes) -> tuple[bool, str]:
    """Returns (is_clean, reason). is_clean=True means safe to proceed."""
    try:
        sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        sock.connect((settings.clamd_host, settings.clamd_port))
        sock.sendall(b"zINSTREAM\0")
        chunk_size = len(file_bytes)
        sock.sendall(chunk_size.to_bytes(4, byteorder="big"))
        sock.sendall(file_bytes)
        sock.sendall((0).to_bytes(4, byteorder="big"))
        result = sock.recv(1024).decode().strip()
        sock.close()
        if "OK" in result:
            return True, "clean"
        else:
            return False, result
    except Exception as e:
        # If ClamAV is unreachable, fail safe — reject the upload
        return False, f"scan_error: {str(e)}"


Step 7 — Storage service (Backblaze B2)
services/storage.py:


import boto3
from botocore.client import Config
from app.config import settings

def get_b2_client():
    return boto3.client(
        "s3",
        endpoint_url=settings.b2_endpoint_url,
        aws_access_key_id=settings.b2_key_id,
        aws_secret_access_key=settings.b2_application_key,
        config=Config(signature_version="s3v4"),
    )

def upload_to_tmp(file_bytes: bytes, filename: str, content_type: str) -> str:
    """Upload to tmp/ prefix before virus scan passes."""
    client = get_b2_client()
    key = f"tmp/{filename}"
    client.put_object(
        Bucket=settings.b2_bucket_name,
        Key=key,
        Body=file_bytes,
        ContentType=content_type,
        ServerSideEncryption="AES256",
    )
    return key

def move_to_final(tmp_key: str, final_key: str):
    """Move from tmp/ to docs/ after scan passes."""
    client = get_b2_client()
    client.copy_object(
        Bucket=settings.b2_bucket_name,
        CopySource={"Bucket": settings.b2_bucket_name, "Key": tmp_key},
        Key=final_key,
        ServerSideEncryption="AES256",
    )
    client.delete_object(Bucket=settings.b2_bucket_name, Key=tmp_key)

def get_presigned_url(key: str, expires_in: int = 900) -> str:
    """15-minute presigned URL for document access."""
    client = get_b2_client()
    return client.generate_presigned_url(
        "get_object",
        Params={"Bucket": settings.b2_bucket_name, "Key": key},
        ExpiresIn=expires_in,
    )

workers/extraction_task.py:

import asyncio, uuid
from workers.celery_app import celery_app
from app.database import SessionLocal
from app.models.document import Document, DocumentStatus
from app.services.virus_scan import scan_bytes
from app.services.extraction import extract_from_image
from app.services.storage import move_to_final, get_b2_client
from app.config import settings

@celery_app.task(bind=True, max_retries=2)
def process_document(self, document_id: str):
    db = SessionLocal()
    try:
        doc = db.query(Document).filter(Document.id == document_id).first()
        if not doc:
            return

        # 1. Download from tmp
        client = get_b2_client()
        response = client.get_object(
            Bucket=settings.b2_bucket_name, Key=doc.s3_tmp_key
        )
        file_bytes = response["Body"].read()

        # 2. Virus scan
        doc.status = DocumentStatus.scanning
        db.commit()
        is_clean, reason = scan_bytes(file_bytes)
        if not is_clean:
            doc.status = DocumentStatus.failed
            db.commit()
            # Delete tmp file — never keep infected files
            client.delete_object(
                Bucket=settings.b2_bucket_name, Key=doc.s3_tmp_key
            )
            return

        # 3. Extract with Claude
        doc.status = DocumentStatus.extracting
        db.commit()
        extracted = asyncio.run(
            extract_from_image(file_bytes, doc.mime_type)
        )
        doc.extracted_data = extracted

        # 4. Move to final encrypted prefix
        final_key = f"docs/{doc.owner_id}/{document_id}"
        move_to_final(doc.s3_tmp_key, final_key)
        doc.s3_key = final_key
        doc.s3_tmp_key = None

        # 5. Mark ready for user review
        doc.status = DocumentStatus.pending_review
        db.commit()

    except Exception as exc:
        doc.status = DocumentStatus.failed
        db.commit()
        raise self.retry(exc=exc, countdown=30)
    finally:
        db.close()

Step 9 — Upload router
routers/documents.py:

import uuid
from fastapi import APIRouter, UploadFile, File, Depends, HTTPException
from sqlalchemy.orm import Session
from app.database import get_db
from app.models.document import Document, DocumentStatus
from app.models.user import User
from app.services.storage import upload_to_tmp
from app.middleware.auth_middleware import get_current_user
from workers.extraction_task import process_document

router = APIRouter(prefix="/documents", tags=["documents"])

ALLOWED_TYPES = {"image/jpeg", "image/png", "application/pdf"}
MAX_SIZE_BYTES = 20 * 1024 * 1024  # 20MB

@router.post("/upload")
async def upload_document(
    file: UploadFile = File(...),
    patient_id: str = None,
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    if file.content_type not in ALLOWED_TYPES:
        raise HTTPException(400, "Only JPEG, PNG, and PDF files are accepted")
    
    file_bytes = await file.read()
    if len(file_bytes) > MAX_SIZE_BYTES:
        raise HTTPException(400, "File exceeds 20MB limit")

    doc_id = str(uuid.uuid4())
    tmp_key = upload_to_tmp(
        file_bytes,
        f"{doc_id}_{file.filename}",
        file.content_type,
    )

    doc = Document(
        id=doc_id,
        owner_id=current_user.id,
        patient_id=patient_id,
        original_filename=file.filename,
        s3_tmp_key=tmp_key,
        mime_type=file.content_type,
        status=DocumentStatus.uploading,
    )
    db.add(doc)
    db.commit()

    # Kick off async pipeline
    process_document.delay(doc_id)

    return {"document_id": doc_id, "status": "processing"}

@router.get("/{document_id}/status")
def get_document_status(
    document_id: str,
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    doc = db.query(Document).filter(
        Document.id == document_id,
        Document.owner_id == current_user.id,
    ).first()
    if not doc:
        raise HTTPException(404, "Document not found")
    return {
        "status": doc.status,
        "extracted_data": doc.extracted_data if doc.status == "pending_review" else None,
    }

import * as SecureStore from "expo-secure-store";

const API_BASE = process.env.EXPO_PUBLIC_API_URL; // set in .env

const api = axios.create({
  baseURL: API_BASE,
  timeout: 30000,
});

// Attach JWT on every request
api.interceptors.request.use(async (config) => {
  const token = await SecureStore.getItemAsync("sanarch_jwt");
  if (token) {
    config.headers.Authorization = `Bearer ${token}`;
  }
  return config;
});

// Auto-refresh on 401
api.interceptors.response.use(
  (res) => res,
  async (error) => {
    if (error.response?.status === 401) {
      await SecureStore.deleteItemAsync("sanarch_jwt");
      // navigate to login — handle via your auth store
    }
    return Promise.reject(error);
  }
);

export default api;

import auth from "@react-native-firebase/auth";
import * as SecureStore from "expo-secure-store";
import api from "./api";

export async function sendOTP(phoneNumber: string) {
  // phoneNumber must be in E.164 format: +919876543210
  return await auth().signInWithPhoneNumber(phoneNumber);
}

export async function verifyOTPAndLogin(
  confirmation: any,
  otp: string
): Promise<void> {
  const credential = await confirmation.confirm(otp);
  const firebaseToken = await credential.user.getIdToken();

  // Exchange Firebase token for your own JWT
  const response = await api.post("/auth/verify-firebase", {
    firebase_token: firebaseToken,
  });

  await SecureStore.setItemAsync("sanarch_jwt", response.data.access_token);
}

export async function logout() {
  await auth().signOut();
  await SecureStore.deleteItemAsync("sanarch_jwt");
}

import { useState } from "react";
import { View, Text, Pressable, ActivityIndicator } from "react-native";
import * as DocumentPicker from "expo-document-picker";
import * as ImagePicker from "expo-image-picker";
import api from "@/services/api";

export default function UploadScreen() {
  const [status, setStatus] = useState
    "idle" | "uploading" | "processing" | "review" | "error"
  >("idle");
  const [extractedData, setExtractedData] = useState(null);
  const [documentId, setDocumentId] = useState<string | null>(null);

  async function uploadFile(uri: string, mimeType: string, name: string) {
    setStatus("uploading");
    const formData = new FormData();
    formData.append("file", { uri, type: mimeType, name } as any);

    try {
      const res = await api.post("/documents/upload", formData, {
        headers: { "Content-Type": "multipart/form-data" },
      });
      setDocumentId(res.data.document_id);
      setStatus("processing");
      pollForResult(res.data.document_id);
    } catch {
      setStatus("error");
    }
  }

  function pollForResult(docId: string) {
    const interval = setInterval(async () => {
      const res = await api.get(`/documents/${docId}/status`);
      if (res.data.status === "pending_review") {
        setExtractedData(res.data.extracted_data);
        setStatus("review");
        clearInterval(interval);
      } else if (res.data.status === "failed") {
        setStatus("error");
        clearInterval(interval);
      }
    }, 2000); // poll every 2 seconds
  }

  async function pickPDF() {
    const result = await DocumentPicker.getDocumentAsync({
      type: "application/pdf",
    });
    if (!result.canceled) {
      const f = result.assets[0];
      uploadFile(f.uri, "application/pdf", f.name);
    }
  }

  async function takePhoto() {
    const result = await ImagePicker.launchCameraAsync({
      mediaTypes: ImagePicker.MediaTypeOptions.Images,
      quality: 0.9,
    });
    if (!result.canceled) {
      const f = result.assets[0];
      uploadFile(f.uri, "image/jpeg", "capture.jpg");
    }
  }

  if (status === "processing") {
    return (
      <View style={{ flex: 1, alignItems: "center", justifyContent: "center" }}>
        <ActivityIndicator size="large" color="#1D4A2A" />
        <Text style={{ marginTop: 16, color: "#666" }}>
          Scanning and extracting data...
        </Text>
      </View>
    );
  }

  // Render review UI with extractedData when status === 'review'
  // Render upload options when status === 'idle'
  return (
    <View style={{ flex: 1, padding: 24 }}>
      <Pressable onPress={takePhoto}>
        <Text>Take Photo</Text>
      </Pressable>
      <Pressable onPress={pickPDF}>
        <Text>Upload PDF</Text>
      </Pressable>
    </View>
  );
}
